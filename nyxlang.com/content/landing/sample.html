enum Result { Ok(int), Err(String) } fn parse_port(text: String) -> Result { let n: int = string_to_int(text) if n < 1 or n > 65535 { return Result.Err("out of range: ${text}") } return Result.Ok(n) } fn bind_port(text: String) -> Result { let port: int = parse_port(text)? // ? hands the Err back to the caller return Result.Ok(port) } fn main() { match bind_port("8080") { Result.Ok(port) => print("listening on ${port}"), Result.Err(why) => print("config error: ${why}") } }
